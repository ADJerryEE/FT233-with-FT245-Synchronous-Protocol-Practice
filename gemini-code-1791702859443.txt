# 1. Define the 60 MHz clock from FT233HP CLKOUT (t1 = 16.67 ns)
create_clock -period 16.670 -name clk_60mhz [get_ports i_clk_60mhz]

# 2. Input Delay for RXF# (t4: Min = 1.4 ns, Max = 9.0 ns) and TXE# (t11: Min = 1.0 ns, Max = 9.0 ns)
set_input_delay -clock clk_60mhz -max 9.000 [get_ports {i_rxf_l i_txe_l}]
set_input_delay -clock clk_60mhz -min 1.000 [get_ports {i_rxf_l i_txe_l}]

# 3. Input Delay for Read DATA (t5: Min = 2.4 ns, Max = 9.0 ns)
set_input_delay -clock clk_60mhz -max 9.000 [get_ports {io_data[*]}]
set_input_delay -clock clk_60mhz -min 2.400 [get_ports {io_data[*]}]

# 4. Output Delay for OE#, WR#, Write DATA (t7, t14, t12 Setup = 8.0 ns, Hold = 0.0 ns)
set_output_delay -clock clk_60mhz -max 8.000 [get_ports {o_oe_l o_wr_l io_data[*]}]
set_output_delay -clock clk_60mhz -min 0.000 [get_ports {o_oe_l o_wr_l io_data[*]}]

# 5. Output Delay for RD# (t9 Setup = 7.5 ns, t10 Hold = 0.0 ns)
set_output_delay -clock clk_60mhz -max 7.500 [get_ports o_rd_l]
set_output_delay -clock clk_60mhz -min 0.000 [get_ports o_rd_l]
